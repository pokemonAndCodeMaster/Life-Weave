0107 class LogNormalNLL(nn.Module):
0108     '''
0109     Negative log-likelihood loss function for log-normal distribution.
0110     '''
0111     def __init__(self, small_eps=small_eps):
0112         super(LogNormalNLL, self).__init__()
0113         self.log2pi = torch.log(torch.tensor(2*3.1415926535897932384626433832795))
0114         self.small_eps = small_eps
0115 
0116     def forward(self, out, s):
0117         mu = out[0]
0118         log_var = out[1]
0119         log_s = torch.log(torch.clamp(s, min=self.small_eps))
0120         nll = 0.5 * (self.log2pi + log_var + (log_s-mu)**2 / torch.exp(log_var)) + log_s
0121         loss = nll.mean()
0122         return loss

0125 class SmoothLogNormalNLL(nn.Module):
0126     '''
0127     Negative log-likelihood loss function for log-normal distribution with smoothness regularization.
0128     '''
0129     def __init__(self, beta=5., small_eps=small_eps):
0130         super(SmoothLogNormalNLL, self).__init__()
0131         self.beta = beta
0132         self.log2pi = torch.log(torch.tensor(2*3.1415926535897932384626433832795))
0133         self.small_eps = small_eps
0134 
0135     def kl_divergence(self, mu1, log_var1, mu2, log_var2):
0136         return 0.5 * (log_var2 - log_var1 + (torch.exp(log_var1)+(mu1-mu2)**2) / torch.exp(log_var2) - 1)
0137     
0138     def js_divergence(self, mu1, log_var1, mu2, log_var2):
0139         mu = 0.5 * (mu1 + mu2)
0140         var = 0.5 * (torch.exp(log_var1) + torch.exp(log_var2))
0141         log_var = torch.log(torch.clamp(var, min=self.small_eps))
0142         kl1 = self.kl_divergence(mu1, log_var1, mu, log_var)
0143         kl2 = self.kl_divergence(mu2, log_var2, mu, log_var)
0144         return 0.5 * (kl1 + kl2)
0145 
0146     def forward(self, out, s, inducing_out):
0147         mu = out[0]
0148         log_var = out[1]
0149         log_s = torch.log(torch.clamp(s, min=self.small_eps))
0150         nll = 0.5 * (self.log2pi + log_var + (log_s-mu)**2 / torch.exp(log_var)) + log_s
0151 
0152         mu_prime, log_var_prime = inducing_out
0153         js_divergence = self.js_divergence(mu, log_var, mu_prime, log_var_prime)
0154         loss = nll.mean() + self.beta*js_divergence.mean()
0155         return loss

0011 class TSEncoder(nn.Module):
0012     '''
0013     This encoder extracts the time series features into 5 representations,
0014     within each the features are designed to **interact** with each other.
0015     '''
0016     def __init__(self, input_dims=4, output_dims=64):
0017         super(TSEncoder, self).__init__()
0018         self.dropout = nn.Dropout(0.2)
0019         self.batch_norm = nn.BatchNorm1d(input_dims)
0020         self.lstm = nn.LSTM(input_dims, output_dims, num_layers=1, batch_first=True, bidirectional=False)
0021 
0022     def forward(self, x): # x: (batch_size, 25, feature_dims=4)
0023         x = torch.flip(x, [1]) # reverse time series to encode the latest time step first
0024         x_bn = self.batch_norm(x.permute(0, 2, 1)) # batch_norm to normalise the features
0025         x_bn = x_bn.permute(0, 2, 1) # (batch_size, feature_dims=4, 25) -> (batch_size, 25, feature_dims=4)
0026         x_bn = self.dropout(x_bn) # dropout to break time steps and avoid positional leakage
0027         output, _ = self.lstm(x_bn) # output: (batch_size, 25, hidden_dims=64)
0028         out = output[:, 4::5, :] # each encode the passed 0.5, 1, 1.5, 2, 2.5 seconds
0029         return out #(batch_size, 5, 64)

0101 class BNRF(nn.Module):
0102     '''
0103     Batch-Norm + Random Feature (BNRF)
0104     This module first attaches an orthogonal Gaussian random feature that is deterministic for a latent representation,
0105     and then applies batch normalisation to the concatenated latent representations inculing random features.
0106     The attached random featurs are with no gradients to regularise the latent space.
0107     '''
0108     def __init__(self, seq_len, additional_dim=None):
0109         super(BNRF, self).__init__()
0110         self.seq_len = seq_len
0111         if additional_dim is None:
0112             self.additional_dim = 2
0113         else:
0114             self.additional_dim = additional_dim
0115         self.batch_norm1d = nn.BatchNorm1d(seq_len+self.additional_dim)
0116         self.register_buffer('projector', # register a buffer to store the orthogonal random feature
0117                              torch.ones((self.additional_dim, self.seq_len), requires_grad=False)*float('inf'))
0118 
0119     @staticmethod
0120     def _make_orthogonal_rows(m, n, device):
0121         # full random Gaussian matrix
0122         G = torch.randn(n, n, device=device, requires_grad=False) # (seq_len, seq_len)
0123         # QR decomposition -> orthonormal columns (so rows of Q^T are orthonormal)
0124         Q, _ = torch.linalg.qr(G, mode='reduced')          # Q: seq_len×seq_len
0125         Q = Q[:m].contiguous()                             # keep first `additional_dim` rows of Q^T
0126                                                            # (additional_dim, seq_len)
0127 
0128         # scaling factors for each row of Q^T
0129         chi_samples = torch.randn(m, n, device=device).norm(dim=1, keepdim=True) # (additional_dim, seq_len)
0130         return (chi_samples / (n**0.5)) * Q  # (additional_dim, seq_len)
0131     
0132     def forward(self, x): # x: (batch_size, seq_len, latent_dims=64)
0133         if self.projector.isinf().all():
0134             self.projector = self._make_orthogonal_rows(self.additional_dim, self.seq_len, x.device)
0135 
0136         # random_feature = torch.einsum('as,bsd->bad', self.projector, x) 
0137         random_feature = torch.matmul(x.transpose(1,2), self.projector.transpose(0,1)).transpose(1,2) # (batch_size, additional_dim, latent_dims=64)
0138         
0139         latent = self.batch_norm1d(torch.cat([x, random_feature], dim=1)) # (batch_size, seq_len+additional_dim, latent_dims=64)
0140         return latent

0143 class AttentionBlock(nn.Module):
0144     '''
0145     This defines a block for the self-attention mechanism.
0146     '''
0147     def __init__(self, input_dims):
0148         super(AttentionBlock, self).__init__()
0149         self.linear_q = nn.Sequential(
0150             nn.Linear(input_dims, input_dims),
0151         )
0152         self.linear_k = nn.Sequential(
0153             nn.Linear(input_dims, input_dims),
0154         )
0155         self.linear_v = nn.Sequential(
0156             nn.Linear(input_dims, input_dims),
0157         )
0158         self.dropout = nn.Dropout(0.2)
0159         self.layer_norm = nn.LayerNorm(input_dims)
0160 
0161     def forward(self, x): # x: (batch_size, seq_len, input_dims)
0162         queries = self.linear_q(x)
0163         keys = self.linear_k(x)
0164         values = self.linear_v(x)
0165 
0166         scores = torch.matmul(queries, keys.transpose(-2, -1)) / (values.size(-1) ** 0.5)
0167         attention = torch.nn.functional.softmax(scores, dim=-1) # (batch_size, seq_len, seq_len)
0168         attention = self.dropout(attention)
0169         attended = torch.matmul(attention, values) # (batch_size, seq_len, output_dims)
0170 
0171         attended = self.layer_norm(attended + x)
0172         return attended, attention

0175 class FeedForwardBlock(nn.Module):
0176     '''
0177     This defines a block for the feed-forward network.
0178     The output dimensions can be either the same as the input dimensions or larger.
0179     If the output dimensions are larger, the number needs to be divisible by the input dimensions
0180     to allow for residual connection.
0181     '''
0182     def __init__(self, input_dims, output_dims):
0183         super(FeedForwardBlock, self).__init__()
0184         self.input_dims = input_dims
0185         self.output_dims = output_dims
0186         self.mlp = nn.Sequential(
0187             nn.Linear(input_dims, output_dims),
0188             nn.Dropout(0.2),
0189             nn.GELU(),
0190             nn.Linear(output_dims, output_dims)
0191         )
0192         self.layer_norm = nn.LayerNorm(output_dims)
0193 
0194     def forward(self, x): # x: (batch_size, seq_len, input_dims)
0195         if self.input_dims == self.output_dims:
0196             residual = x
0197         else: # output_dims needs to be larger than and divisible by input_dims
0198             residual = torch.repeat_interleave(x, self.output_dims//self.input_dims, dim=-1)
0199         out = self.mlp(x)
0200         out = self.layer_norm(out + residual)
0201         return out

0311                                                            # attention_matrices: {block_i: [batch_size, seq_len, seq_len]}

0029 def partial_auc(fpr, tpr, min_tpr=0.80, resolution=1000, normalize=True):
0030     '''
0031     Integrate the ROC curve only where TPR >= min_tpr
0032     '''
0033     fpr = np.asarray(fpr, dtype=float)
0034     tpr = np.asarray(tpr, dtype=float)
0035 
0036     # sort the points by FPR
0037     order = np.argsort(fpr)
0038     fpr, tpr = fpr[order], tpr[order]
0039 
0040     # interpolate onto a dense, uniform FPR grid
0041     virtual_fpr = np.linspace(small_eps, 1 - small_eps, resolution)
0042     interp_tpr = np.interp(virtual_fpr, fpr, tpr)
0043 
0044     # keep only the part above the TPR floor
0045     mask = interp_tpr >= min_tpr
0046     if not mask.any():
0047         return 0.0
0048     
0049     area = np.trapz(interp_tpr[mask] - min_tpr, virtual_fpr[mask])
0050     if normalize:
0051         area /= (1.0 - min_tpr)            # theoretical max area
0052     return area

0055 def get_auprc(recall, precision, resolution=1000):

0057     Area under the Precision-Recall curve
0058     '''

0060     precision = np.asarray(precision, dtype=float)

0085 def _median_ci_bounds(x: pd.Series, alpha: float = 0.01):
0086     """
0087     Exact (1 - alpha) CI for the population median via order statistics.
0088     Uses Binomial(n, 0.5) quantiles (SciPy) — stable for large n.
0089     Returns (lower_bound, upper_bound).
0090     """
0091     arr = np.asarray(x.dropna())
0092     n = arr.size
0093     if n == 0:
0094         return (np.nan, np.nan)
0095 
0096     arr.sort()
0097 
0098     # Binomial quantiles for p = 0.5
0099     kL = int(binom.ppf(alpha / 2.0, n, 0.5))          # in [0, n]
0100     kU = int(binom.ppf(1.0 - alpha / 2.0, n, 0.5))    # in [0, n]
0101 
0102     # Map to 0-based order-statistic indices:
0103     # Lower = kL, Upper = kU - 1  (clipped to [0, n-1])
0104     lo_idx = max(0, min(kL, n - 1))
0105     hi_idx = max(0, min(kU - 1, n - 1))
0106 
0107     return (arr[lo_idx], arr[hi_idx])
0108 
0109 def get_low_CI(TTI: pd.Series, alpha: float = 0.05) -> float:
0110     """Lower bound of the exact (1 - alpha) CI for the median."""
0111     return _median_ci_bounds(TTI, alpha=alpha)[0]
0112 
0113 def get_up_CI(TTI: pd.Series, alpha: float = 0.05) -> float:
0114     """Upper bound of the exact (1 - alpha) CI for the median."""
0115     return _median_ci_bounds(TTI, alpha=alpha)[1]

0118 def get_time(warning, f1=None, cutoff=1.5):
0119     '''
0120     Median time-to-impact (TTI) and the proportion with TTI ≥ cutoff.
0121     '''

0124 
0125     if f1 is None:
0126         tp, fp, tn, fn = get_statistics(warning, return_statistics=False)
0127         f1 = tp / np.maximum(small_eps, tp + 0.5*(fp + fn))
0128     median_tti = w[w['TTI']<10].groupby('threshold')['TTI'].median()
0129     tti_q25 = w[w['TTI']<10].groupby('threshold')['TTI'].quantile(0.25)
0130     tti_q75 = w[w['TTI']<10].groupby('threshold')['TTI'].quantile(0.75)

0137                 tti_lowCI.loc[f1.idxmax()],
0138                 tti_upCI.loc[f1.idxmax()])

0060 def lognormal_pdf(x, mu, sigma, rescale=True):
0061     p = 1/x/np.sqrt(2*np.pi)/sigma*np.exp(-1/2*(np.log(x)-mu)**2/sigma**2)
0062     if rescale:
0063         mode = np.exp(mu-sigma**2)

0168 def is_target_recorded(danger, pre_danger, target_id, indicator):
0169     '''
0170     The conflicting target was not marked in the SHRP2 data, and it's possible the target was not recorded in
0171     the data. We thus need to filter out those invalid cases. 
0172     Later the method evaluation will assess missed detection based on [the target in danger] and assess false
0173     detection based on [other surrounding vehicles in the pre-danger period]. 
0174     To remain those eval-use data unintervened, this filtering is based on [the target in the pre-danger period] 
0175     and [other surrounding vehicles in the danger period]. It is thus assumed that the conflicting target is less
0176     risky in the pre-danger period than within the danger period. 
0177     More specifically,
0178     1) if the selected target has no data before the danger period, a filtering is not possible so we 
0179        skip the event, better being conservative than misleading;
0180     2) for the selected target, the 25th, 50th, and 75th percentiles of the risk indicated by the indicator values
0181        should be less in the pre-danger period than in the danger period, otherwise, the real conflicting target
0182        might be missed.
0183     * The use of 25th, 50th, and 75th percentiles (rather than e.g., min, mean, max) intends for a more robust 
0184       comparison and to avoid potential influence of outliers.
0185     '''
0186     danger = danger.loc[[target_id]]
0187     pre_danger = pre_danger[pre_danger['target_id']==target_id]
0188 
0189     # 1) No data before the danger period
0190     if len(pre_danger)<1:
0191         target_not_recorded = True
0192         percentiles_pre_danger = [np.nan, np.nan, np.nan]
0193         percentiles_danger = [np.nan, np.nan, np.nan]
0194     else:
0195         target_not_recorded = False
0196         # To avoid error in returning percentiles, replace inf with a large value and -inf with a small value
0197         pre_danger_positive_inf = np.isinf(pre_danger[indicator])&(pre_danger[indicator]>0)
0198         pre_danger_negative_inf = np.isinf(pre_danger[indicator])&(pre_danger[indicator]<0)
